```python
"""KI-Assistent repository — zeitplan entries + processed sources."""

import json
from datetime import date, datetime, time
from typing import Any, Optional

from psycopg2.extras import RealDictCursor

from src.core.db.pool import get_connection


# ─── Entries ────────────────────────────────────────────────────────────────

def get_zeitplan_entries(module_id: int, user_id: int) -> list[dict]:
    with get_connection() as conn:
        with conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(
                """
                SELECT id, type, title, description, due_date, due_time,
                       is_optional, is_checked, checked_at, origin,
                       source_ids, raw_quote, created_at, updated_at
                  FROM kia_zeitplan_entries
                 WHERE module_id = %s AND user_id = %s
                 ORDER BY due_date NULLS LAST, due_time NULLS LAST, id
                """,
                (module_id, user_id),
            )
            return [dict(r) for r in cur.fetchall()]


def get_zeitplan_entry(entry_id: int, user_id: int) -> Optional[dict]:
    with get_connection() as conn:
        with conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(
                """
                SELECT id, module_id, user_id, type, title, description,
                       due_date, due_time, is_optional, is_checked, checked_at,
                       origin, source_ids, raw_quote, created_at, updated_at
                  FROM kia_zeitplan_entries
                 WHERE id = %s AND user_id = %s
                """,
                (entry_id, user_id),
            )
            row = cur.fetchone()
            return dict(row) if row else None


def create_zeitplan_entry(
    module_id: int,
    user_id: int,
    *,
    type: str,
    title: str,
    description: Optional[str],
    due_date: Optional[date],
    due_time: Optional[time],
    is_optional: bool,
    origin: str,
    source_ids: list[str],
    raw_quote: Optional[str],
) -> dict:
    with get_connection() as conn:
        with conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(
                """
                INSERT INTO kia_zeitplan_entries
                    (module_id, user_id, type, title, description,
                     due_date, due_time, is_optional, origin,
                     source_ids, raw_quote)
                VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
                RETURNING id, type, title, description, due_date, due_time,
                          is_optional, is_checked, checked_at, origin,
                          source_ids, raw_quote, created_at, updated_at
                """,
                (
                    module_id, user_id, type, title, description,
                    due_date, due_time, is_optional, origin,
                    json.dumps(source_ids), raw_quote,
                ),
            )
            row = dict(cur.fetchone())
        conn.commit()
        return row


def update_zeitplan_entry(
    entry_id: int,
    user_id: int,
    *,
    type: Optional[str] = None,
    title: Optional[str] = None,
    description: Optional[str] = None,
    due_date: Optional[date] = None,
    due_time: Optional[time] = None,
    is_optional: Optional[bool] = None,
    is_checked: Optional[bool] = None,
    clear_date: bool = False,
    clear_time: bool = False,
) -> Optional[dict]:
    """Partial update. Pass only fields to change.
    clear_date/clear_time set the field to NULL even if due_date=None is also falsy.
    is_checked auto-updates checked_at.
    """
    sets: list[str] = []
    params: list[Any] = []

    if type is not None:
        sets.append("type = %s")
        params.append(type)
    if title is not None:
        sets.append("title = %s")
        params.append(title)
    if description is not None:
        sets.append("description = %s")
        params.append(description)
    if clear_date:
        sets.append("due_date = NULL")
    elif due_date is not None:
        sets.append("due_date = %s")
        params.append(due_date)
    if clear_time:
        sets.append("due_time = NULL")
    elif due_time is not None:
        sets.append("due_time = %s")
        params.append(due_time)
    if is_optional is not None:
        sets.append("is_optional = %s")
        params.append(is_optional)
    if is_checked is not None:
        sets.append("is_checked = %s")
        params.append(is_checked)
        sets.append("checked_at = %s")
        params.append(datetime.utcnow() if is_checked else None)

    if not sets:
        return get_zeitplan_entry(entry_id, user_id)

    sets.append("updated_at = NOW()")
    sql = (
        "UPDATE kia_zeitplan_entries SET "
        + ", ".join(sets)
        + " WHERE id = %s AND user_id = %s "
        + "RETURNING id, type, title, description, due_date, due_time, "
          "is_optional, is_checked, checked_at, origin, source_ids, raw_quote, "
          "created_at, updated_at"
    )
    params.extend([entry_id, user_id])

    with get_connection() as conn:
        with conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(sql, params)
            row = cur.fetchone()
        conn.commit()
        return dict(row) if row else None


def delete_zeitplan_entry(entry_id: int, user_id: int) -> bool:
    with get_connection() as conn:
        with conn.cursor() as cur:
            cur.execute(
                "DELETE FROM kia_zeitplan_entries WHERE id = %s AND user_id = %s",
                (entry_id, user_id),
            )
            deleted = cur.rowcount > 0
        conn.commit()
        return deleted


def delete_all_zeitplan_entries(module_id: int, user_id: int) -> int:
    with get_connection() as conn:
        with conn.cursor() as cur:
            cur.execute(
                "DELETE FROM kia_zeitplan_entries WHERE module_id = %s AND user_id = %s",
                (module_id, user_id),
            )
            count = cur.rowcount
            cur.execute(
                "DELETE FROM kia_zeitplan_processed_sources WHERE module_id = %s AND user_id = %s",
                (module_id, user_id),
            )
        conn.commit()
        return count


# ─── Processed Sources Tracker ───────────────────────────────────────────────

def get_processed_source_ids(module_id: int, user_id: int) -> list[int]:
    with get_connection() as conn:
        with conn.cursor() as cur:
            cur.execute(
                """
                SELECT source_id
                  FROM kia_zeitplan_processed_sources
                 WHERE module_id = %s AND user_id = %s
                """,
                (module_id, user_id),
            )
            return [r[0] for r in cur.fetchall()]


def mark_source_processed(module_id: int, user_id: int, source_id: int) -> None:
    with get_connection() as conn:
        with conn.cursor() as cur:
            cur.execute(
                """
                INSERT INTO kia_zeitplan_processed_sources (module_id, user_id, source_id)
                VALUES (%s, %s, %s)
                ON CONFLICT DO NOTHING
                """,
                (module_id, user_id, source_id),
            )
        conn.commit()
```

```python
"""KI-Assistent routes — zeitplan."""

import json
import re
from datetime import date, datetime, time
from typing import Optional

from fastapi import APIRouter, Depends, HTTPException
from loguru import logger

from langfuse.decorators import langfuse_context, observe

from src import database as db
from src.dependencies import get_current_user, get_kisski_client
from src.features.ki_assistent import schemas
from src.tool_prompts import KIA_ZEITPLAN_PROMPT

router = APIRouter(tags=["ki-assistent"])


# ─── Helpers ────────────────────────────────────────────────────────────────

def _require_module_access(module_uuid: str, user_id: int) -> dict:
    module = db.get_kia_module(module_uuid)
    if not module:
        raise HTTPException(status_code=404, detail="Modul nicht gefunden")
    is_owner = module["user_id"] == user_id
    is_enrolled = db.is_student_enrolled(module["id"], user_id)
    if not is_owner and not is_enrolled:
        raise HTTPException(status_code=403, detail="Kein Zugriff")
    return module


def _parse_date(s: Optional[str]) -> Optional[date]:
    if not s:
        return None
    try:
        return date.fromisoformat(s)
    except ValueError:
        raise HTTPException(status_code=400, detail=f"Ungültiges Datum: {s}")


def _parse_time(s: Optional[str]) -> Optional[time]:
    if not s:
        return None
    try:
        parts = s.split(":")
        return time(int(parts[0]), int(parts[1]))
    except (ValueError, IndexError):
        raise HTTPException(status_code=400, detail=f"Ungültige Uhrzeit: {s}")


def _serialize_entry(row: dict) -> dict:
    """DB-row → JSON-safe dict for API response."""
    return {
        "id": row["id"],
        "type": row["type"],
        "title": row["title"],
        "description": row.get("description"),
        "due_date": row["due_date"].isoformat() if row.get("due_date") else None,
        "due_time": row["due_time"].strftime("%H:%M") if row.get("due_time") else None,
        "is_optional": row["is_optional"],
        "is_checked": row["is_checked"],
        "checked_at": row["checked_at"].isoformat() + "Z" if row.get("checked_at") else None,
        "origin": row["origin"],
        "source_ids": row.get("source_ids") or [],
        "raw_quote": row.get("raw_quote"),
        "created_at": row["created_at"].isoformat() + "Z" if row.get("created_at") else None,
        "updated_at": row["updated_at"].isoformat() + "Z" if row.get("updated_at") else None,
    }


# ─── GET ────────────────────────────────────────────────────────────────────

@router.get("/ki-assistent/modules/{module_uuid}/zeitplan")
async def get_zeitplan(
    module_uuid: str,
    current_user: dict = Depends(get_current_user),
):
    user_id = current_user["user_id"]
    module = _require_module_access(module_uuid, user_id)

    entries = db.get_zeitplan_entries(module["id"], user_id)
    processed_ids = db.get_processed_source_ids(module["id"], user_id)
    processed_uuids: list[str] = []
    for sid in processed_ids:
        src = db.get_kia_source_by_id(sid) if hasattr(db, "get_kia_source_by_id") else None
        if src:
            processed_uuids.append(src["uuid"])

    return {
        "entries": [_serialize_entry(e) for e in entries],
        "processed_source_uuids": processed_uuids,
    }


# ─── GENERATE ───────────────────────────────────────────────────────────────

@observe(name="kia-zeitplan-extract")
def _extract_for_source(kisski, source_text: str) -> list[dict]:
    """Einmaliger LLM-Call, gibt Array von rohen Entry-Dicts zurück.
    Bei Fehler oder invalidem JSON: Retry 1x, dann Exception.
    """
    def _call() -> list[dict]:
        response = kisski._client.with_options(timeout=60.0).chat.completions.create(
            model=kisski._model,
            messages=[
                {"role": "system", "content": KIA_ZEITPLAN_PROMPT},
                {"role": "user", "content": source_text},
            ],
            temperature=0.1,
            max_tokens=2000,
            **(kisski._extra_body and {"extra_body": kisski._extra_body} or {}),
        )
        raw = (response.choices[0].message.content or "").strip()
        # strip markdown fences if model added them
        raw = re.sub(r"^```(?:json)?\s*|\s*```$", "", raw.strip(), flags=re.MULTILINE)
        return json.loads(raw)

    try:
        result = _call()
    except (json.JSONDecodeError, Exception) as e:
        logger.warning("Zeitplan-Extract erster Versuch fehlgeschlagen: {}. Retry.", e)
        result = _call()  # one retry, raises on second failure

    if not isinstance(result, list):
        raise ValueError(f"LLM lieferte kein Array: {type(result)}")
    return result


def _entry_from_llm(raw: dict) -> dict:
    """Wandelt LLM-Rohoutput in create-zeitplan-entry kwargs."""
    type_ = raw.get("type")
    if type_ not in ("deadline", "termin"):
        type_ = "termin"
    due_date = _parse_date(raw.get("due_date"))
    due_time = _parse_time(raw.get("due_time"))
    return {
        "type": type_,
        "title": str(raw.get("title") or "Ohne Titel")[:200],
        "description": (raw.get("description") or None),
        "due_date": due_date,
        "due_time": due_time,
        "is_optional": bool(raw.get("is_optional") or False),
        "raw_quote": (raw.get("raw_quote") or None),
    }


def _load_source_text(source: dict) -> str:
    """Lade Volltext einer Quelle für LLM-Verarbeitung.
    Nutzt kia_chunks (bereits extrahierter Text) und konkateniert in Seiten-Reihenfolge.
    """
    chunks = db.get_kia_chunks_for_source(source["id"]) if hasattr(db, "get_kia_chunks_for_source") else []
    if not chunks:
        return ""
    ordered = sorted(chunks, key=lambda c: (c.get("page_number") or 0, c.get("id", 0)))
    parts: list[str] = []
    for c in ordered:
        page = c.get("page_number")
        prefix = f"[S. {page}] " if page else ""
        parts.append(prefix + (c.get("content") or "").strip())
    return "\n\n".join(parts)[:50000]  # safety cap to avoid blowing context


@router.post("/ki-assistent/modules/{module_uuid}/zeitplan/generate")
async def generate_zeitplan(
    module_uuid: str,
    body: schemas.ZeitplanGenerateBody,
    current_user: dict = Depends(get_current_user),
    kisski=Depends(get_kisski_client),
):
    user_id = current_user["user_id"]
    module = _require_module_access(module_uuid, user_id)

    # Resolve UUIDs → source rows
    sources: list[dict] = []
    for suuid in body.source_ids:
        src = db.get_kia_source(suuid)
        if not src or src["module_id"] != module["id"]:
            continue
        sources.append(src)

    # Filter: skip already-processed
    already_processed = set(db.get_processed_source_ids(module["id"], user_id))
    pending = [s for s in sources if s["id"] not in already_processed]

    if not pending:
        return {"added": [], "skipped_already_processed": len(sources)}

    added_entries: list[dict] = []
    for src in pending:
        text = _load_source_text(src)
        if not text.strip():
            db.mark_source_processed(module["id"], user_id, src["id"])
            continue
        try:
            raw_entries = _extract_for_source(kisski, text)
        except Exception as e:
            logger.error("Zeitplan-Extract fehlgeschlagen fuer source_id={}: {}", src["id"], e)
            # Quelle NICHT als processed markieren, damit Retry moeglich ist
            raise HTTPException(status_code=502, detail=f"LLM-Extraktion fehlgeschlagen: {e}")

        for raw in raw_entries:
            kwargs = _entry_from_llm(raw)
            row = db.create_zeitplan_entry(
                module["id"],
                user_id,
                origin="llm",
                source_ids=[src["uuid"]],
                **kwargs,
            )
            added_entries.append(_serialize_entry(row))

        db.mark_source_processed(module["id"], user_id, src["id"])

    return {
        "added": added_entries,
        "skipped_already_processed": len(sources) - len(pending),
    }
```

```typescript
function parseYMD(s: string): Date {
  const [y, m, d] = s.split("-").map(Number);
  return new Date(y, m - 1, d);
}

function sortByDate(entries: ZeitplanEntry[]): ZeitplanEntry[] {
  return [...entries].sort((a, b) => {
    const da = a.due_date ? parseYMD(a.due_date).getTime() : 0;
    const db = b.due_date ? parseYMD(b.due_date).getTime() : 0;
    if (da !== db) return da - db;
    const ta = a.due_time || "";
    const tb = b.due_time || "";
    return ta.localeCompare(tb);
  });
}

function formatDay(dateStr: string): { day: string; month: string } {
  const d = parseYMD(dateStr);
  const months = ["Jan", "Feb", "Mär", "Apr", "Mai", "Jun", "Jul", "Aug", "Sep", "Okt", "Nov", "Dez"];
  return { day: String(d.getDate()).padStart(2, "0"), month: months[d.getMonth()] };
}

function isToday(dateStr: string, today: Date): boolean {
  const d = parseYMD(dateStr);
  return (
    d.getFullYear() === today.getFullYear() &&
    d.getMonth() === today.getMonth() &&
    d.getDate() === today.getDate()
  );
}

  const d = parseYMD(dateStr);

      )}
```

```typescript
      <div className="flex-1 overflow-y-auto px-3 py-2">
        {loading && (
          <p className="text-xs py-6 text-center" style={{ color: "var(--muted)" }}>Lade Zeitplan...</p>
        )}
        {error && (
          <p className="text-xs py-6 text-center" style={{ color: "#B91C1C" }}>{error}</p>
        )}
        {!loading && !error && entries.length === 0 && (
          <div className="flex flex-col items-center gap-3 py-10 text-center">
            <CalendarBlank size={40} weight="light" style={{ color: "var(--muted)" }} />
            <p className="text-xs" style={{ color: "var(--muted)" }}>
              Noch keine Einträge. Generiere aus deinen Quellen oder erstelle manuelle Einträge.
            </p>
          </div>
        )}
        {!loading && entries.length > 0 && (
          <>
            <ZeitplanTimeline
              entries={datedEntries}
              editMode={editMode}
              onToggleChecked={handleToggleChecked}
              onEdit={setEditing}
              onDelete={handleDelete}
            />
            <ZeitplanOpenSection
              entries={openEntries}
              editMode={editMode}
              onToggleChecked={handleToggleChecked}
              onEdit={setEditing}
              onDelete={handleDelete}
            />
          </>
        )}
      </div>

      {/* Modals */}
      {showGenerate && (
        <ZeitplanGenerateModal
          sources={sources}
          processedUuids={processedUuids}
          onCancel={() => setShowGenerate(false)}
          onGenerate={handleGenerate}
        />
      )}
      {generating && (
        <div className="fixed inset-0 z-50 flex items-center justify-center" style={{ backgroundColor: "rgba(0,0,0,0.4)" }}>
          <div className="rounded-xl p-5 text-sm" style={{ backgroundColor: "var(--bg)", color: "var(--text)" }}>
            Einträge werden extrahiert...
          </div>
        </div>
      )}
      {creatingManual && (
        <ZeitplanEntryEditModal
          title="Neuer Eintrag"
          submitLabel="Anlegen"
          onCancel={() => setCreatingManual(false)}
          onSubmit={handleCreateManual}
        />
      )}
      {editing && (
        <ZeitplanEntryEditModal
          title="Eintrag bearbeiten"
          submitLabel="Speichern"
          initial={editing}
          onCancel={() => setEditing(null)}
          onSubmit={handleEditSubmit}
        />
      )}
      {confirmDeleteAll && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4" style={{ backgroundColor: "rgba(0,0,0,0.4)" }} onClick={() => setConfirmDeleteAll(false)}>
          <div className="rounded-2xl w-full max-w-sm p-5" style={{ backgroundColor: "var(--bg)" }} onClick={(e) => e.stopPropagation()}>
            <h3 className="text-base font-bold mb-2" style={{ color: "var(--text)" }}>Zeitplan komplett löschen?</h3>
            <p className="text-xs" style={{ color: "var(--muted)" }}>
              Alle Einträge werden gelöscht und alle Quellen werden wieder generierbar. Diese Aktion kann nicht rückgängig gemacht werden.
            </p>
            <div className="mt-4 flex justify-end gap-2">
              <button
                type="button"
                onClick={() => setConfirmDeleteAll(false)}
                className="px-4 py-2 rounded-lg text-xs font-semibold"
                style={{ color: "var(--muted)", backgroundColor: "var(--surface-dim)" }}
              >
                Abbrechen
              </button>
              <button
                type="button"
                onClick={handleDeleteAll}
                className="px-4 py-2 rounded-lg text-xs font-semibold"
                style={{ color: "#fff", backgroundColor: "#B91C1C" }}
              >
                Alles löschen
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
```
